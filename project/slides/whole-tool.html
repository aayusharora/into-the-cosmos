def sky_conditions(lat: float = 26.73, lon: float = 88.40) -> dict:
"""How cloudy will it be tonight at this place, hour by hour.
0 is a clear sky; above about 40 there is no point going outside."""
 
raw = _get("https://api.open-meteo.com/v1/forecast?" + urlencode(
{"latitude": lat, "longitude": lon,
"hourly": "cloud_cover", "forecast_days": 2}))
 
h = json.loads(raw)["hourly"]
ev = [(t[11:16], c) for t, c in zip(h["time"], h["cloud_cover"])
if 18 <= int(t[11:13]) <= 23]
 
return {"evening": dict(ev),
"average_cloud_pct": round(sum(c for _, c in ev) / len(ev))}
